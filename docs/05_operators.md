# Arithmetic Operators
x = 10
y = 3
print(x + y)
print(x - y)
print(x * y)
print(x / y)
print(x % y)
print(x ** y)
print(x // y)

# Comparison Operators
print(x == y)
print(x != y)
print(x > y)
print(x < y)
print(x >= y)
print(x <= y)

# Logical Operators
a = True
b = False
print(a and b)
print(a or b)
print(not a)

# Assignment Operators
z = 5
z += 3
print(z)
z -= 2
print(z)
z *= 4
print(z)
z /= 3
print(z)

# Membership Operators
nums = [1, 2, 3, 4, 5]
print(3 in nums)
print(6 not in nums)

# Identity Operators
m = [1, 2, 3]
n = [1, 2, 3]
print(m is n)
print(m is not n)
