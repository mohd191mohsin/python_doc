# Integer
a = 10
print(type(a))

# Float
b = 3.14
print(type(b))

# String
c = "Hello"
print(type(c))

# Boolean
d = True
print(type(d))

# List
e = [1, 2, 3, 4]
print(type(e))

# Tuple
f = (1, 2, 3, 4)
print(type(f))

# Set
g = {1, 2, 3, 4}
print(type(g))

# Dictionary
h = {"name": "Alice", "age": 25}
print(type(h))
